# Pista 3. Esqueleto

```kotlin
private val json = Json { ? = true }

fun fetchBooks(): ApiResult<List<BookDto>> {
    val request = Request.Builder().url("${baseUrl.trimEnd('/')}/books").build()
    client.newCall(request).execute().use { response ->
        if (!response.?) return ApiResult.HttpError(response.?)
        return try {
            ApiResult.Success(json.decodeFromString<List<BookDto>>(response.body.string()))
        } catch (e: ?) {
            ApiResult.ParseError
        }
    }
}
```
